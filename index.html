// Aapka Live Render API URL const RENDER_API_URL = 'https://shagun-api.onrender.com'; // Form submit hone par ye function chalega async function createStockGift(giftDetails) { try { const response = await fetch(`${RENDER_API_URL}/api/v1/gifts/create`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ sender_phone: giftDetails.senderPhone, // Sender ka Phone sender_name: giftDetails.senderName, // Sender ka Name recipient_phone: giftDetails.recipientPhone, // Recipient ka Phone recipient_name: giftDetails.recipientName, // Recipient ka Name stock_symbol: giftDetails.stockSymbol, // e.g. 'TATAMOTORS' gift_amount_inr: giftDetails.amount, // e.g. 2100 occasion: giftDetails.occasion, // e.g. 'WEDDING' message: giftDetails.message // Personal Message }) }); const data = await response.json(); if (data.success) { alert(`Gift Order Created! Claim Link: ${data.claim_url}`); console.log('Order Data:', data.order); } else { alert('Error: ' + data.error); } } catch (err) { console.error('Network Error:', err); } }
